function auto_invis_touch(prt, hrp, hl, hldata, bp, bg)
    local bv = prt:FindFirstChild("igrab")
    local po = prt:FindFirstChild("PartOwner")
    local sv = prt:FindFirstChild("whograb")
    local con
    if not sv then
        sv = create_sv(prt, "whograb", "")
        con = prt.ChildAdded:Connect(function(obj)
            if obj.Name == "PartOwner" then
                sv.Value = obj.Value
            end
        end)
    end
    function auto()
        set_hl(hl, "FillColor", hldata, 3)
        local save = hrp.CFrame
        local t = false
        local last_pos = prt.Position
        task.spawn(function()
            while (not po or po.Value ~= O.myname) and task.wait(0.1) do
                if not prt.Parent or not hrp.Parent then
                    break
                end
                if check_hum(hrp.Parent.Humanoid) and last_pos ~= prt.Position then
                    last_pos = prt.Position
                end
            end
        end)
        while (not po or po.Value ~= O.myname) and task.wait() do
            if not prt.Parent or not hrp.Parent then
                break
            end
            if check_hum(hrp.Parent.Humanoid) then
                if check_mag2(prt, hrp) then
                    set_hl(hl, "FillColor", hldata, 4)
                    hrp.CFrame = prt.CFrame + ((prt.Position - last_pos) * O.me:GetNetworkPing() * 100)
                    t = true
                    invis_touch(prt)
                else
                    invis_touch(prt)
                end
            else
                set_hl(hl, "FillColor", hldata, 5)
            end
            po = prt:FindFirstChild("PartOwner")
        end
        if t then
            hrp.CFrame = save
        end
        set_hl(hl, "FillColor", hldata, 1)
    end
    if sv.Value ~= O.me.Name then
        auto()
    end
    if bv then
        set_hl(hl, "FillColor", hldata, 2)
        while not (prt:FindFirstChild("PartOwner")) do
            task.wait()
        end
        bp.MaxForce = N.V.nv3
        bg.MaxTorque = N.V.nv3
        while bv.Parent do
            task.wait()
        end
        bp.Position = prt.Position
        bg.CFrame = prt.CFrame
        bp.MaxForce = N.V.mhv3
        bg.MaxTorque = N.V.mhv3
        set_hl(hl, "FillColor", hldata, 1)
    end
end

function safe_invis_touch(prt, hrp, hl, hldata, bp, bg)

        if check_limbs(prt.Name) then

        else
            local po = prt:FindFirstChild("PartOwner")
            local save = hrp.CFrame
            local t = false
            local last_pos = prt.Position
            task.spawn(function()
                while t3 and task.wait(0.1) do
                    if not prt.Parent or not hrp.Parent then
                        break
                    end
                    if check_hum(hrp.Parent.Humanoid) and last_pos ~= prt.Position then
                        last_pos = prt.Position
                    end
                end
            end)
            if not po or po.Value ~= O.myname then
                while (not po or po.Value ~= O.myname) and task.wait() do
                    if not prt.Parent or not hrp.Parent then
                        break
                    end

    character = Players.LocalPlayer.Character
    if not toggleBringEnabled or not character or not character:FindFirstChild("Head") or not part:IsA("BasePart") then
        return
    end
    if part.Anchored or part:IsDescendantOf(character) then
        return
    end
    if part.Name == "Torso" or part.Name == "Head" or part.Name == "Right Arm" or part.Name == "Left Arm" or part.Name ==
        "Right Leg," or part.Name == "Left Leg" or part.Name == "HumanoidRootPart" then
        return
    end
    if not part:FindFirstChildOfClass("BodyPosition") then
        ForceInstance = Instance.new("BodyPosition")
        ForceInstance.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        ForceInstance.Position = Vector3.new(0, R_raaa, 0)
        ForceInstance.P = 6000000000000
        ForceInstance.D = 6000000
        ForceInstance.Parent = part
    end
end

    if blobman then
        trackedBlobman = blobman

                trackedBlobman = nil
            end

                pcall(function()

                wait(0.01)
            end
        end)

                startTracking(child)
                return
            end
        end
    end

        end
    end)

function autoMountBlobman(enabled)
    autoMountEnabled = enabled
    if enabled then
        onPlayerRespawn()
        monitorSpawnedInToys()
        if not trackedBlobman then
            findAndTrackBlobman()
        end
    else
        if spawnedInToysConnection then
            spawnedInToysConnection:Disconnect()
            spawnedInToysConnection = nil
        end
        trackedBlobman = nil
    end
end

function updateBodyMovers(primaryPart)
    for _, group in ipairs(compiledGroups) do
        if group.primaryPart and group.primaryPart == primaryPart then
            for _, data in ipairs(group.group) do
                local bodyPosition = data.part:FindFirstChild("BodyPosition")
                local bodyGyro = data.part:FindFirstChild("BodyGyro")
                if bodyPosition then
                    bodyPosition.Position = (primaryPart.CFrame * data.offset).Position
                end
                if bodyGyro then
                    bodyGyro.CFrame = primaryPart.CFrame * data.offset
                end
            end
        end
    end
end
function cleanupCompiledGroups()
    for _, groupData in ipairs(compiledGroups) do
        for _, data in ipairs(groupData.group) do
            if data.part then
                if data.part:FindFirstChild("BodyPosition") then
                    data.part.BodyPosition:Destroy()
                end
                if data.part:FindFirstChild("BodyGyro") then
                    data.part.BodyGyro:Destroy()
                end
            end
        end
        if groupData.primaryPart and groupData.primaryPart.Parent then
            local highlight = groupData.primaryPart:FindFirstChild("Highlight") or
                                  groupData.primaryPart.Parent:FindFirstChild("Highlight")
            if highlight then
                highlight:Destroy()
            end
        end
    end
    cleanupConnections(compileConnections)
    cleanupConnections(renderSteppedConnections)
    compiledGroups = {}
end

            end
        end)

    end
end

do
    local HttpService = game:GetService("HttpService")

    local CFG_FILE = "SuperStrengthConfig.json"

    local function fsReady()
        return type(readfile) == "function" and type(writefile) == "function"
    end

    local function fileExists(path)
        if type(isfile) == "function" then
            local ok, v = pcall(isfile, path)
            return ok and v or false
        end
        local ok = pcall(function() return readfile(path) end)
        return ok
    end

    local function loadCfg()
        if not fsReady() then return {} end
        if not fileExists(CFG_FILE) then return {} end
        local ok, decoded = pcall(function()
            local raw = readfile(CFG_FILE)
            if type(raw) ~= "string" or raw == "" then return {} end
            local obj = HttpService:JSONDecode(raw)
            return type(obj) == "table" and obj or {}
        end)
        return ok and decoded or {}
    end

    local function saveCfg(cfg)
        if not fsReady() then return end
        pcall(function()
            writefile(CFG_FILE, HttpService:JSONEncode(cfg))
        end)
    end

    local cfg = loadCfg()
    _G.SuperStrength = (cfg.SuperStrength ~= nil) and (cfg.SuperStrength and true or false) or (_G.SuperStrength and true or false) or false
    _G.Strength = tonumber(cfg.Strength) or tonumber(_G.Strength) or 1000

        local _, hrp, hum = getCharInfo()

            pcall(ragdollRemote.FireServer, ragdollRemote, hrp, 0)

        if not model or model.Name ~= "CreatureBlobman" then return end
        for _, v in ipairs(model:GetDescendants()) do
            if v:IsA("BasePart") then v.Anchored = true end
        end
    end

        task.spawn(function()
            local _, _, hum = getCharInfo()
            if not hum then return end
            hum.Sit = true
            task.wait(duration)
            if hum.Parent and hum.Health > 0 then
                hum.Sit = false
                task.wait(0.05)
                pcall(hum.ChangeState, hum, Enum.HumanoidStateType.GettingUp)
            end
        end)
    end

    local function createSystem()
        local state = {
            enabled    = false,
            targetToy  = nil,
            currentToy = nil,
            savedPos   = nil,
        }

        local function runSequence(child)
            if not child or not child.Parent then return end
            if child.Name ~= state.targetToy then return end
            state.currentToy = child

            local _, hrp, hum = getCharInfo()
            if not (hrp and hum) then return end

            if state.targetToy == "CreatureBlobman" then
                anchorBlobman(child)
            end

            local seat
            local t = tick()
            repeat
                seat = child:FindFirstChildWhichIsA("VehicleSeat", true)
                if seat then break end
                RunService.Heartbeat:Wait()
            until tick() - t > 1

            if not (seat and hum and hum.Parent) then return end

            pcall(seat.Sit, seat, hum)

            local deadline = tick() + ANTI_DURATION
            while tick() < deadline and state.enabled do
                pcall(function()
                    if ragdollRemote and hrp.Parent and hum.Parent then
                        for _ = 1, 10 do
                            ragdollRemote:FireServer(hrp, 0)
                        end
                        hum:ChangeState(Enum.HumanoidStateType.Jumping)
                    end
                end)
                RunService.Heartbeat:Wait()
            end

            local primary = child.PrimaryPart
                or child:FindFirstChild("HumanoidRootPart", true)
                or child:FindFirstChild("Part", true)

            if primary and primary.Parent and state.enabled then
                pcall(function()
                    if primary.SetNetworkOwner then
                        primary:SetNetworkOwner(localPlayer)
                    end
                    if typeof(sethiddenproperty) == "function" then
                        sethiddenproperty(primary, "NetworkIsSleeping", false)
                    end
                end)
            end

            if state.savedPos and hrp.Parent then
                hrp.CFrame = state.savedPos
            end

            task.wait()

            if seat and seat.Parent then
                pcall(seat.Destroy, seat)
            end

            local _, _, humPost = getCharInfo()
            if humPost and humPost.Parent then
                pcall(function()
                    humPost.Sit = false
                    task.wait(0.05)
                    humPost:ChangeState(Enum.HumanoidStateType.GettingUp)
                end)
            end
        end

        local function spawnToy()
            if not state.enabled then return end
            if not toysFolder or not toysFolder.Parent then return end

            pcall(function()
                local before = {}
                for _, c in ipairs(toysFolder:GetChildren()) do before[c] = true end

                spawnRemote:InvokeServer(
                    state.targetToy,
                    CFrame.new(SPAWN_POS),
                    Vector3.new(0, -15.716, 0)
                )

                local newToy
                local t = tick()
                repeat
                    for _, c in ipairs(toysFolder:GetChildren()) do
                        if not before[c] and c.Name == state.targetToy then
                            newToy = c
                            break
                        end
                    end
                    if newToy then break end
                    RunService.Heartbeat:Wait()
                until tick() - t > 2

                if newToy and state.enabled then
                    runSequence(newToy)
                end
            end)
        end

        local function startAutoLoop()
            task.spawn(function()
                while state.enabled do
                    local toy = state.currentToy
                    if not (toy and toy.Parent) then
                        spawnToy()
                    end
                    RunService.Heartbeat:Wait()
                end
            end)
        end

        function state.start(toyName)
            state.enabled   = true
            state.targetToy = toyName

            local _, hrp = getCharInfo()
            if hrp then state.savedPos = hrp.CFrame end

            startAutoLoop()

            state.currentToy = nil
            spawnToy()
        end

        function state.stop(sitDuration, waitTime)
            state.enabled = false

            task.wait(waitTime)

            clearRagdoll()
            pcall(function()
                if state.currentToy and state.currentToy.Parent then
                    destroyRemote:FireServer(state.currentToy)
                end
            end)
            state.currentToy = nil
            state.savedPos   = nil
            sit(sitDuration)
        end

        localPlayer.CharacterAdded:Connect(function(char)
            if not state.enabled then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
                or char:FindFirstChild("Torso")
                or char:WaitForChild("HumanoidRootPart", 5)
            if hrp then state.savedPos = hrp.CFrame end
            state.currentToy = nil
            spawnToy()
        end)

        return state
    end

function SetAntiBlobman(enabled)
    if not AntiCheatStates then
        AntiCheatStates = {}
    end
    
    local LocalPlayer = game.Players.LocalPlayer
    
    local function processPart(part)
        if part:IsA("Weld") or part:IsA("AlignOrientation") then
            local partName = part.Name
            local invalid = partName == "LeftWeld" or partName == "RightWeld" or 
                           partName == "LeftAlignOrientation" or partName == "RightAlignOrientation"
            
            if invalid then
                local parentModel = part.Parent
                if parentModel and parentModel.Parent then
                    local grandParent = parentModel.Parent
                    if grandParent:IsA("Model") or grandParent:IsA("BasePart") then
                        local owner = nil
                        
                        if grandParent:IsA("Model") then
                            local humanoid = grandParent:FindFirstChildOfClass("Humanoid")
                            if humanoid then
                                owner = game.Players:GetPlayerFromCharacter(grandParent)
                            end
                        end
                        
                        if owner ~= LocalPlayer then
                            if part:IsA("Weld") then
                                part.Enabled = not enabled
                            elseif part:IsA("AlignOrientation") then
                                part.Enabled = not enabled
                            end
                        end
                    end
                end
            end
        end
    end
    
    if AntiCheatStates.blobConn then
        AntiCheatStates.blobConn:Disconnect()
        AntiCheatStates.blobConn = nil
    end
    
    if enabled then
        for _, part in pairs(workspace:GetDescendants()) do
            pcall(function()
                processPart(part)
            end)
        end
        
        AntiCheatStates.blobConn = workspace.DescendantAdded:Connect(function(part)
            task.wait()
            pcall(function()
                processPart(part)
            end)
        end)
    else
        for _, part in pairs(workspace:GetDescendants()) do
            pcall(function()
                if (part:IsA("Weld") or part:IsA("AlignOrientation")) and 
                   (part.Name == "LeftWeld" or part.Name == "RightWeld" or 
                    part.Name == "LeftAlignOrientation" or part.Name == "RightAlignOrientation") then
                    part.Enabled = true
                end
            end)
        end
    end
end

                    local sign = v:FindFirstChild("PlotSign")

                    if owners then
                        for _, b in pairs(owners:GetChildren()) do
                            if b.Value == plr.Name then
                                local folder = Workspace.PlotItems:FindFirstChild(v.Name)
                                if folder then return true, folder end
                            end
                        end
                    end
                end
                return false
            end

            function StickKunai(kunai)
                if not kunai or not kunai:FindFirstChild("StickyPart") then return end
                local currentHRP = getHRP()
                if not currentHRP then return end

                if kunai:FindFirstChild("SoundPart") then
                    if not kunai.SoundPart:FindFirstChild("PartOwner") or kunai.SoundPart.PartOwner.Value ~= plr.Name then
                        setOwner:FireServer(kunai.SoundPart, kunai.SoundPart.CFrame)
                    end
                end

                local firePart = currentHRP:FindFirstChild("FirePlayerPart") or currentHRP:WaitForChild("FirePlayerPart", 5)
                if firePart then
                    stickyEvent:FireServer(
                        kunai.StickyPart,
                        firePart,
                        CFrame.new(0,0,0) * CFrame.Angles(0, math.rad(90), math.rad(90))
                    )
                end

                for _, obj in pairs(kunai:GetChildren()) do
                    if obj:IsA("BasePart") then
                        obj.CanTouch = false
                        obj.CanCollide = false
                        obj.CanQuery = false
                        obj.Transparency = 1
                    end
                end
            end

            local function SpawnToy(name)
                local t = tick()
                while not canSpawn.Value do
                    if not _G.ShurikenAntiKick or tick() - t > 5 then return nil end

                end

                local currentHRP = getHRP()

                end

                local boolik, house = CheckForHome()
                local inv = Workspace:FindFirstChild(plr.Name.."SpawnedInToys")

                if boolik and house then
                    return house:WaitForChild(name, 2)
                elseif not Workspace.PlotItems.PlayersInPlots:FindFirstChild(plr.Name) and inv then
                    return inv:WaitForChild(name, 2)
                end
                return nil
            end


            local function IsAttached(kunai)
                if not kunai or not kunai.Parent then return false end
                local sp = kunai:FindFirstChild("StickyPart")
                if not sp then return false end
                -- CanTouch が true = まだ/もう貼り付いていない
                if sp.CanTouch == true then return false end
                -- StickyWeld が存在し Part1 が設定されているか確認
                local weld = sp:FindFirstChild("StickyWeld")
                if not weld or not weld.Part1 then return false end
                -- プレイヤーとの距離チェック（15以上で外れと判断）
                local hrp = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
                if hrp and (hrp.Position - sp.Position).Magnitude >= 15 then return false end
                return true
            end

            while _G.ShurikenAntiKick do
                task.wait(0.005)

                if not plr.Character
                    or not plr.Character:FindFirstChild("Humanoid")
                    or plr.Character.Humanoid.Health <= 0
                then
                    continue
                end

                local inv = Workspace:FindFirstChild(plr.Name.."SpawnedInToys")
                local kunai = inv and inv:FindFirstChild("NinjaShuriken")

                if Workspace.PlotItems.PlayersInPlots:FindFirstChild(plr.Name) then
                    local boolik, house = CheckForHome()
                    if boolik and house and Workspace.Plots:FindFirstChild(house.Name) then
                        local sign = Workspace.Plots[house.Name]:FindFirstChild("PlotSign")
                        if sign and sign.ThisPlotsOwners.Value.TimeRemainingNum.Value > 89 then
                            kunai = SpawnToy("NinjaShuriken")
                            if kunai == nil then continue end
                            kunai.Name = "AntiKick"
                            StickKunai(kunai)
                        end
                    end
                end

                if not kunai then
                    if Workspace.PlotItems.PlayersInPlots:FindFirstChild(plr.Name) then continue end
                    kunai = SpawnToy("NinjaShuriken")
                    if kunai == nil then continue end
                    kunai.Name = "AntiKick"
                    if not kunai then continue end
                end

                
                local detached = false
                local connections = {}

                local stickyPart = kunai:FindFirstChild("StickyPart")
                if stickyPart then
                    -- CanTouch が変わった瞬間に検知
                    connections[1] = stickyPart:GetPropertyChangedSignal("CanTouch"):Connect(function()
                        if stickyPart.CanTouch == true then
                            detached = true
                        end
                    end)
                    -- StickyWeld が削除された瞬間に検知
                    connections[2] = stickyPart.ChildRemoved:Connect(function(child)
                        if child.Name == "StickyWeld" then
                            detached = true
                        end
                    end)
                end

                -- kunai が消えた瞬間に検知
                connections[3] = kunai.AncestryChanged:Connect(function()
                    if not kunai.Parent then
                        detached = true
                    end
                end)

                -- まだ付いていなければ即装着
                if not IsAttached(kunai) then
                    StickKunai(kunai)
                    kunai.Name = "AntiKick"
                end

                
                repeat
                    task.wait(0.05)

                    if not IsAttached(kunai) then
                        detached = true
                    end
                until not _G.ShurikenAntiKick
                    or detached
                    or not kunai
                    or not plr.Character
                    or not plr.Character:FindFirstChild("HumanoidRootPart")

                
                for _, conn in pairs(connections) do
                    pcall(function() conn:Disconnect() end)
                end


                if detached
                    or not kunai
                    or not kunai:FindFirstChild("StickyPart")
                    or not plr.Character
                    or not plr.Character:FindFirstChild("HumanoidRootPart")
                then
                    ClearKunai()
                
                    task.wait(0.06)
                end
            end
        end)
    else
        _G.ShurikenAntiKick = false
        ClearKunai()
    end
end

local function GetRepresentativePart(character, partKey)
    if not character then return nil end
    local function findAny(names)
        for _, n in ipairs(names) do
            local p = character:FindFirstChild(n)
            if p and p:IsA("BasePart") then return p end
        end
        return nil
    end
    if partKey == "Head" then return findAny({"Head"})
    elseif partKey == "HumanoidRootPart" then return findAny({"HumanoidRootPart","Torso","UpperTorso","LowerTorso"})
    elseif partKey == "Left Arm" then return findAny({"LeftHand","LeftUpperArm","Left Arm","LeftLowerArm"})
    elseif partKey == "Right Arm" then return findAny({"RightHand","RightUpperArm","Right Arm","RightLowerArm"})
    elseif partKey == "Left Leg" then return findAny({"LeftFoot","LeftLowerLeg","Left Leg","LeftUpperLeg"})
    elseif partKey == "Right Leg" then return findAny({"RightFoot","RightLowerLeg","Right Leg","RightUpperLeg"})
    end
    return nil
end

            end

                player = player,
                root = targetRoot,
                char = char,
                distance = (targetRoot.Position - root.Position).Magnitude
            }
        end
    end

    table.sort(targets, function(a, b)
        return a.distance < b.distance
    end)

    return targets

                            angle = angle + angularSpeed * dt
                            local newPos = Vector3.new(
                                savedTargetPos.X + math.cos(angle) * RADIUS,
                                fixedY,
                                savedTargetPos.Z + math.sin(angle) * RADIUS
                            )
                            root.CFrame = CFrame.new(newPos, Vector3.new(savedTargetPos.X, fixedY, savedTargetPos.Z))
                        end
                    end)

                    blobKick(AuraBlobman, targetRoot, "Left")

                    kickMoving = false
                    task.cancel(randomMoveThread)

                    setNoclip(false)

                    removeAllForces(char)

                    if player and player.UserId then _G.__KickCooldown[player.UserId] = {
                        time = os.clock(),
                        char = char
                    } end

                    task.wait(0.03)
                end
            end

            task.wait(0.03)
        end
    end)
end

function stopKickAll()
    _G.__OrionKickAll = false

    if kickLoop then
        task.cancel(kickLoop)
        kickLoop = nil
    end

    kickAllShouldRemount = false
    kickAllLastSitState = false
    toggleOnPosition = nil
    _G.__KickCooldown = {}
    destroyBlobman()
end

local function isOwnedByPlayer(plotModel, plr)
    if not (plotModel and plr) then return false end
    local auid = plotModel:GetAttribute("OwnerUserId") or plotModel:GetAttribute("OwnerId")
    if typeof(auid) == "number" and auid == plr.UserId then return true end
    local aname = plotModel:GetAttribute("OwnerName")
    if typeof(aname) == "string" and aname == plr.Name then return true end
    local owner = plotModel:FindFirstChild("Owner") or plotModel:FindFirstChild("PlotOwner")

        if owner:IsA("ObjectValue") and owner.Value == plr then return true end
        if owner:IsA("IntValue") and owner.Value == plr.UserId then return true end

    local ownerId = plotModel:FindFirstChild("OwnerUserId") or plotModel:FindFirstChild("OwnerId")

       and math.abs(localPos.Z) <= half.Z

        local BP = Part:FindFirstChildOfClass("BodyPosition") or Instance.new("BodyPosition")
        local BG = Part:FindFirstChildOfClass("BodyGyro") or Instance.new("BodyGyro")
        BP.P = 15000; BP.D = 200; BP.MaxForce = Vector3.new(1, 1, 1) * 1e10; BP.Parent = Part
        BG.P = 15000; BG.D = 200; BG.MaxTorque = Vector3.new(1, 1, 1) * 1e10; BG.Parent = Part
        return BG, BP
    end

    local function BuildWings()
        CleanupWings()
        local folders = GetToyFolders()
        local allSpawnedToys = {}
        for _, folder in ipairs(folders) do
            for _, x in ipairs(folder:GetDescendants()) do
                if x:IsA("Model") and x.Name == SelectedItem then
                    table.insert(allSpawnedToys, x)
                end
            end
        end
        if #allSpawnedToys == 0 then return false end
        for i = 1, 2 do

            for x = 1, WingLength do 
                local p = Instance.new("Part")
                p.CanCollide = false; p.Anchored = true; p.Transparency = 1; p.Size = Vector3.new(4,1,4); p.Parent = workspace
                Segments[#Segments+1] = {Part = p}
            end
            local h = Instance.new("Part")
            h.CanCollide = false; h.Anchored = true; h.Transparency = 1; h.Size = Vector3.new(4,1,4); h.Parent = workspace
            Wings[#Wings+1] = {Handle = h, Segments = Segments, Sync = {}, Reserved = nil}
        end
        for i, v in ipairs(allSpawnedToys) do
            local Side = (i <= (#allSpawnedToys/2)) and 1 or 2
            local Pallet = v:FindFirstChild("SoundPart") or v:FindFirstChild("Handle") or v:FindFirstChildWhichIsA("BasePart")
            if Pallet then
                for _, child in pairs(v:GetChildren()) do if child:IsA("BasePart") then child.CanCollide = false end end
                local BG, BP = SetupPhysics(Pallet)
                if not Wings[Side].Reserved then Wings[Side].Reserved = {BG = BG, BP = BP}
                else table.insert(Wings[Side].Sync, {BG = BG, BP = BP}) end
            end
        end
        return true
    end

            dropRemote = nil

        local function refreshBurger()
            local toysFolder = Workspace:FindFirstChild(toysFolderName)
            if not toysFolder then
                clearBurger()
                return
            end
            local b = toysFolder:FindFirstChild(ToyName)
            if not b or not b.Parent then
                clearBurger()
                return
            end
            burger = b
            holdPart = b:FindFirstChild("HoldPart")
            if not holdPart or not holdPart.Parent then
                clearBurger()
                return
            end
            holdRemote = holdPart:FindFirstChild("HoldItemRemoteFunction")
            dropRemote = holdPart:FindFirstChild("DropItemRemoteFunction")
            if not holdRemote or not dropRemote then
                clearBurger()
                return
            end
            if not holdRemote:IsA("RemoteFunction") or not dropRemote:IsA("RemoteFunction") then
                clearBurger()
            end
        end

        local SpawnRemote = menuToys:WaitForChild("SpawnToyRemoteFunction")

        local function spawnBurger(hrp)
            local now = tick()
            if now - lastSpawnTime < SPAWN_COOLDOWN then return end
            lastSpawnTime = now
            SpawnRemote:InvokeServer(ToyName, hrp.CFrame * CFrame.new(0, 5, 0), Zero)
        end

        local AntiInputLagConn = nil
        local AntiInputLagEnabled = false

        local function setHamburgerAuraEnabled(Value)
            AntiInputLagEnabled = Value

            if not Value then
                if AntiInputLagConn then
                    AntiInputLagConn:Disconnect()
                    AntiInputLagConn = nil
                end
                return
            end

            local plr = localPlayer
            local RunServiceLocal = RunService

            if AntiInputLagConn then
                AntiInputLagConn:Disconnect()
                AntiInputLagConn = nil
            end

            AntiInputLagConn = RunServiceLocal.Heartbeat:Connect(function()
                if not AntiInputLagEnabled then
                    if AntiInputLagConn then
                        AntiInputLagConn:Disconnect()
                        AntiInputLagConn = nil
                    end
                    return
                end

                local char, hrp = getCharAndRoot()
                if not char or not hrp then
                    return
                end

                if not burger or not burger.Parent or not holdPart or not holdPart.Parent or not holdRemote or not dropRemote then
                    refreshBurger()
                    if not burger or not burger.Parent or not holdPart or not holdPart.Parent or not holdRemote or not dropRemote then
                        spawnBurger(hrp)
                        return
                    end
                end

                local holdingValue = holdPart:FindFirstChild("HoldingPlayer")
                local holdingPlayer = holdingValue and holdingValue.Value

                if holdingPlayer and holdingPlayer ~= plr then
                    if burger and burger.Parent then
                        pcall(function()
                            burger:Destroy()
                        end)
                    end
                    clearBurger()
                    return
                end

                local b = burger
                local hrpCF = hrp.CFrame

                for i = 1, BURST_PER_FRAME do
                    if not AntiInputLagEnabled then break end
                    if not b or not b.Parent or not holdPart or not holdPart.Parent or not holdRemote or not dropRemote then
                        clearBurger()
                        break
                    end
                    local ok1 = pcall(function() holdRemote:InvokeServer(b, char) end)
                    local ok2 = pcall(function() dropRemote:InvokeServer(b, hrpCF * DropOffset, Zero) end)
                    if not ok1 or not ok2 then
                        clearBurger()
                        break
                    end
                end
            end)
        end

        local NoClipEnabled = false

        local NoClipCharTracker = nil

        local function disconnectConnections(conns)
            if not conns then return end
            for _, conn in ipairs(conns) do
                if conn then
                    pcall(function()
                        conn:Disconnect()
                    end)
                end

        local function createNoClipTracker(model)
            if not model then return nil end

            local tracker = {
                Model = model,
                Parts = {},
                Original = {},
                Connections = {}
            }

            local function addPart(d)
                if not d:IsA("BasePart") then return end
                if tracker.Parts[d] then return end

                tracker.Parts[d] = true
                tracker.Original[d] = {
                    CanCollide = d.CanCollide,
                    CanQuery = d.CanQuery,
                    CanTouch = d.CanTouch
                }

                d.CanCollide = false
                d.CanQuery = false
                d.CanTouch = false
            end

            local function removePart(d)
                if tracker.Parts[d] then
                    tracker.Parts[d] = nil
                end
            end

            for _, d in ipairs(model:GetDescendants()) do
                addPart(d)
            end

            tracker.Connections[#tracker.Connections + 1] = model.DescendantAdded:Connect(addPart)
            tracker.Connections[#tracker.Connections + 1] = model.DescendantRemoving:Connect(removePart)

            return tracker
        end

        local function applyNoClipTracker(tracker)
            if not tracker or not tracker.Model or not tracker.Model.Parent then return end

            for part in pairs(tracker.Parts) do
                if part and part.Parent then
                    if part.CanCollide or part.CanQuery or part.CanTouch then
                        part.CanCollide = false
                        part.CanQuery = false
                        part.CanTouch = false
                    end
                else
                    tracker.Parts[part] = nil
                end
            end
        end

        local function restoreNoClipTracker(tracker)
            if not tracker then return end

            for part, st in pairs(tracker.Original) do
                if part and part.Parent and typeof(st) == "table" then
                    pcall(function()
                        part.CanCollide = st.CanCollide
                        part.CanQuery = st.CanQuery
                        part.CanTouch = st.CanTouch
                    end)
                end
            end

            disconnectConnections(tracker.Connections)
            tracker.Parts = {}
            tracker.Original = {}
            tracker.Connections = {}
        end

        local function restoreVehicleNoClip()
            restoreNoClipTracker(NoClipVehicleTracker)
            NoClipVehicleTracker = nil
        end

        local function restoreCharNoClip()
            restoreNoClipTracker(NoClipCharTracker)
            NoClipCharTracker = nil
        end

        local function restoreOthersNoClip()
            for model, tracker in pairs(NoClipOtherTrackers) do
                restoreNoClipTracker(tracker)
                NoClipOtherTrackers[model] = nil
            end
            NoClipOtherTrackers = {}
        end

        local function clearSeatCache()
            if NoClipSeatCache then
                disconnectConnections(NoClipSeatCache.Connections)
                NoClipSeatCache = nil
            end
        end

        local function createSeatCache(model)
            if not model then return nil end

            local cache = {
                Model = model,
                Seats = {},
                Connections = {}
            }

            local function addSeat(d)
                if d:IsA("Seat") or d:IsA("VehicleSeat") then
                    cache.Seats[d] = true
                end
            end

            local function removeSeat(d)
                if cache.Seats[d] then
                    cache.Seats[d] = nil
                end
            end

            for _, d in ipairs(model:GetDescendants()) do
                addSeat(d)
            end

            cache.Connections[#cache.Connections + 1] = model.DescendantAdded:Connect(addSeat)
            cache.Connections[#cache.Connections + 1] = model.DescendantRemoving:Connect(removeSeat)

            return cache
        end

        local function ensureVehicleTracker(model)
            if not model then
                restoreVehicleNoClip()
                clearSeatCache()
                return
            end

            if not NoClipVehicleTracker or NoClipVehicleTracker.Model ~= model then
                restoreVehicleNoClip()
                NoClipVehicleTracker = createNoClipTracker(model)
            end

            if not NoClipSeatCache or NoClipSeatCache.Model ~= model then
                clearSeatCache()
                NoClipSeatCache = createSeatCache(model)
            end
        end

        local function ensureCharTracker(model)
            if not model then
                restoreCharNoClip()
                return
            end

            if not NoClipCharTracker or NoClipCharTracker.Model ~= model then
                restoreCharNoClip()
                NoClipCharTracker = createNoClipTracker(model)
            end
        end

        local function updateOtherOccupantsNoClip(vehicleModel)
            if not vehicleModel then
                restoreOthersNoClip()
                return
            end

            if not NoClipSeatCache or NoClipSeatCache.Model ~= vehicleModel then
                clearSeatCache()
                NoClipSeatCache = createSeatCache(vehicleModel)
            end

            local activeModels = {}

            for seat in pairs(NoClipSeatCache.Seats) do
                if seat and seat.Parent then
                    local occ = seat.Occupant
                    local occChar = occ and occ.Parent

                    if occChar and occChar:IsA("Model") and occChar ~= localPlayer.Character then
                        activeModels[occChar] = true

                        local tracker = NoClipOtherTrackers[occChar]
                        if not tracker then
                            tracker = createNoClipTracker(occChar)
                            NoClipOtherTrackers[occChar] = tracker
                        end

                        applyNoClipTracker(tracker)
                    end
                else
                    NoClipSeatCache.Seats[seat] = nil
                end
            end

            for model, tracker in pairs(NoClipOtherTrackers) do
                if not activeModels[model] or not model.Parent then
                    restoreNoClipTracker(tracker)
                    NoClipOtherTrackers[model] = nil
                end
            end
        end

        local function setNoClip(v)
            NoClipEnabled = v

            if not NoClipEnabled then
                restoreVehicleNoClip()
                restoreCharNoClip()
                restoreOthersNoClip()
                clearSeatCache()
            end
        end

            BV.Velocity = Vector3.zero
            BV.Parent = part

            BG.MaxTorque = Vector3.new(1e9, 1e9, 1e9)

            BG.CFrame = part.CFrame
            BG.Parent = part
            TargetPart = part
            VFlyEnabled = true
        end

                if TrainControl or VFlyEnabled or NoClipEnabled then
                    hardStopAll()
                end
                return
            end

            local seatPart = hum.SeatPart

            if TrainControl then
                if InitialRunning then
                    return
                end

                if HadSeatOnce and not seatPart then
                    hardStopAll()
                    return
                end

                if not seatPart then
                    if tick() - TrainEnableAt <= SeatGraceSeconds then
                        if tick() - LastSitAttempt >= SitRetryInterval then
                            LastSitAttempt = tick()
                            sitTrain()
                        end
                        return
                    else
                        hardStopAll()
                        return
                    end
                end

                if not isOnTrainSeat(seatPart) then
                    if tick() - TrainEnableAt <= SeatGraceSeconds then
                        if tick() - LastSitAttempt >= SitRetryInterval then
                            LastSitAttempt = tick()
                            sitTrain()
                        end
                        return
                    else
                        hardStopAll()
                        return
                    end
                end

                local justMounted = false
                if not HadSeatOnce then
                    justMounted = true
                end

                HadSeatOnce = true
                enableCameraNoClip()

                if NoClipEnabled then
                    local vModel = getVehicleModelFromSeat(seatPart)
                    local cModel = localPlayer.Character

                    ensureVehicleTracker(vModel)
                    ensureCharTracker(cModel)

                    if NoClipVehicleTracker then
                        applyNoClipTracker(NoClipVehicleTracker)
                    end

                    if NoClipCharTracker then
                        applyNoClipTracker(NoClipCharTracker)
                    end

                    updateOtherOccupantsNoClip(vModel)
                else
                    restoreVehicleNoClip()
                    restoreCharNoClip()
                    restoreOthersNoClip()
                    clearSeatCache()
                end

                if not VFlyEnabled then
                    startVFlyForSeat(seatPart)
                end
            else
                if VFlyEnabled or NoClipEnabled then
                    hardStopAll()
                end
            end

            if not VFlyEnabled then return end
            if not (TargetPart and TargetPart.Parent and BV and BG) then
                hardStopAll()
                return
            end

            local cam = Workspace.CurrentCamera
            if not cam then return end

            local move = getMoveVector()
            local dir = Vector3.zero

            if move.Magnitude > 0 then
                dir = cam.CFrame.RightVector * move.X + cam.CFrame.LookVector * (-move.Z)
                if dir.Magnitude > 0 then
                    dir = dir.Unit
                end
            end

            local speed = FLY_SPEED * SPEED_MULT
            BV.Velocity = dir * speed
            BG.CFrame = CFrame.lookAt(TargetPart.Position, TargetPart.Position + cam.CFrame.LookVector, cam.CFrame.UpVector)
        end)

        localPlayer.CharacterAdded:Connect(function()
            task.wait(0.5)
            hardStopAll()
        end)
    end)
end

function esp(p, cr)
    if not ESPEnabled then
        return
    end
    local h = cr:WaitForChild("Humanoid")
    local hrp = cr:WaitForChild("HumanoidRootPart")
    local text = Drawing.new("Text")
    text.Visible = false
    text.Center = true
    text.Outline = true
    text.Font = 2
    text.Color = Color3.fromRGB(255, 255, 255)
    text.Size = 17
    local c1, c2, c3
    local function cleanup()
        text.Visible = false
        text:Remove()
        if c1 then
            c1:Disconnect()
            c1 = nil
        end
        if c2 then
            c2:Disconnect()
            c2 = nil
        end
        if c3 then
            c3:Disconnect()
            c3 = nil
        end
    end
    c1 = game:GetService("RunService").RenderStepped:Connect(function()
        if not ESPEnabled then
            cleanup()
            return
        end
        local hrp_pos, hrp_on_screen = workspace.CurrentCamera:WorldToViewportPoint(hrp.Position)
        if hrp_on_screen then
            text.Position = Vector2.new(hrp_pos.X, hrp_pos.Y)
            text.Text = string.format("%s (%.1f m)", p.Name,
                (hrp.Position - workspace.CurrentCamera.CFrame.Position).Magnitude)
            text.Visible = true
        else
            text.Visible = false
        end
    end)
    c2 = cr.AncestryChanged:Connect(function(_, parent)
        if not parent then
            cleanup()
        end
    end)
    c3 = h.HealthChanged:Connect(function(health)
        if health <= 0 then
            cleanup()
        end
    end)
end

function isDescendantOf(target, other)
    local currentParent = target.Parent
    while currentParent do
        if currentParent == other then
            return true
        end
        currentParent = currentParent.Parent
    end
    return false
end

 function getNearestPlayer()
    local nearestPlayer
    local nearestDistance = math.huge
    local playerCharacter = localPlayer.Character
    if not playerCharacter or not playerCharacter:FindFirstChild("HumanoidRootPart") then
        return nil
    end

    for _, player in pairs(Players:GetPlayers()) do
        if player ~= localPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local distance = (playerCharacter.HumanoidRootPart.Position - player.Character.HumanoidRootPart.Position).Magnitude
            if distance < nearestDistance then
                nearestDistance = distance
                nearestPlayer = player
            end
        end
    end

    return nearestPlayer
end

 function cleanupConnections(connectionTable)
    for i = #connectionTable, 1, -1 do
        connectionTable[i]:Disconnect()
        table.remove(connectionTable, i)
    end
end
 function createHighlight(parent)
    local highlight = Instance.new("Highlight")
    highlight.DepthMode = Enum.HighlightDepthMode.Occluded
    highlight.FillTransparency = 1
    highlight.Name = "Highlight"
    highlight.OutlineColor = Color3.new(0, 0, 1)
    highlight.OutlineTransparency = 0.5
    highlight.Parent = parent
    return highlight
end

 function createBodyMovers(part, position, rotation)
    local bodyPosition = Instance.new("BodyPosition")

    bodyGyro.D = 200

end

 function anchorGrab()
    while true do
        pcall(function()
            local grabParts = workspace:FindFirstChild("GrabParts")
            if not grabParts then
                return
            end

            local grabPart = grabParts:FindFirstChild("GrabPart")
            if not grabPart then
                return
            end

            local weldConstraint = grabPart:FindFirstChild("WeldConstraint")
            if not weldConstraint or not weldConstraint.Part1 then
                return
            end

            local parentModel = weldConstraint.Part1.Parent
            if not parentModel then
                return
            end

            local primaryPart = parentModel.PrimaryPart
            if not primaryPart then
                return
            end

            if isDescendantOf(primaryPart, workspace.Map) then
                return
            end

            for _, player in pairs(Players:GetPlayers()) do
                if player.Character and isDescendantOf(primaryPart, player.Character) then
                    return
                end
            end

            if not table.find(anchoredParts, primaryPart) then
                local highlight = createHighlight(primaryPart.Parent)
                table.insert(anchoredParts, primaryPart)

                local connection = primaryPart.Parent.DescendantAdded:Connect(function(descendant)
                    if onPartOwnerAdded then
                        onPartOwnerAdded(descendant, primaryPart)
                    end
                end)
                table.insert(anchoredConnections, connection)
            end

            for _, child in ipairs(primaryPart:GetChildren()) do
                if child:IsA("BodyPosition") or child:IsA("BodyGyro") then
                    child:Destroy()
                end
            end

            while workspace:FindFirstChild("GrabParts") do
                wait()
            end
            createBodyMovers(primaryPart, primaryPart.Position, primaryPart.CFrame)
        end)
        wait()
    end
end

function cleanupAnchoredParts()
    for _, part in ipairs(anchoredParts) do
        if part then
            if part:FindFirstChild("BodyPosition") then
                part.BodyPosition:Destroy()
            end
            if part:FindFirstChild("BodyGyro") then
                part.BodyGyro:Destroy()
            end
            if part.Parent and part.Parent:FindFirstChild("Highlight") then
                part.Parent.Highlight:Destroy()
            end
        end
    end

    cleanupConnections(anchoredConnections)
    anchoredParts = {}
end
 function processPart(part)
    local character = Players.LocalPlayer.Character
    if not toggleBringEnabled or not character or not character:FindFirstChild("Head") or not part:IsA("BasePart") then
        return
    end
    if part.Anchored or part:IsDescendantOf(character) then
        return
    end
    if part.Name == "Torso" or part.Name == "Head" or part.Name == "Right Arm" or part.Name == "Left Arm" or
       part.Name == "Right Leg" or part.Name == "Left Leg" or part.Name == "HumanoidRootPart" then
        return
    end
    if not part:FindFirstChildOfClass("BodyPosition") then
        local ForceInstance = Instance.new("BodyPosition")
        ForceInstance.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        ForceInstance.Position = Vector3.new(0, R_raaa, 0)
        ForceInstance.P = 6000000000000
        ForceInstance.D = 6000000
        ForceInstance.Parent = part
    end
end
